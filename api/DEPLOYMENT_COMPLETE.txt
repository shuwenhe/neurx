💡 使用示例
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  🐍 Python 示例
  ──────────────────────────────────────────────────────────────────────────
  import requests
  
  response = requests.post("http://localhost:8888/v1/chat/completions", json={
      "model": "Qwen2.5-0.5B-Instruct",
      "messages": [{"role": "user", "content": "What is AI?"}],
      "max_tokens": 256
  })
  
  result = response.json()
  print(result["choices"][0]["message"]["content"])

  🌐 JavaScript/Node.js 示例
  ──────────────────────────────────────────────────────────────────────────
  const fetch = require('node-fetch');
  
  const response = await fetch('http://localhost:8888/v1/chat/completions', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      model: 'Qwen2.5-0.5B-Instruct',
      messages: [{role: 'user', content: 'Hello'}],
      max_tokens: 128
    })
  });
  
  const data = await response.json();
  console.log(data.choices[0].message.content);

  🧪 cURL 示例
  ──────────────────────────────────────────────────────────────────────────
  # 健康检查
  curl http://localhost:8888/health | jq .
  
  # 聊天
  curl -X POST http://localhost:8888/v1/chat/completions \
    -H "Content-Type: application/json" \
    -d '{"model": "Qwen2.5-0.5B-Instruct", 
         "messages": [{"role": "user", "content": "Hello"}], 
         "max_tokens": 100}' | jq .

  🐧 Bash 脚本示例
  ──────────────────────────────────────────────────────────────────────────
  #!/bin/bash
  
  API_URL="http://localhost:8888"
  MODEL="Qwen2.5-0.5B-Instruct"
  PROMPT="${1:-What is machine learning?}"
  
  curl -X POST "$API_URL/v1/chat/completions" \
    -H "Content-Type: application/json" \
    -d "{
      \"model\": \"$MODEL\",
      \"messages\": [{\"role\": \"user\", \"content\": \"$PROMPT\"}],
      \"max_tokens\": 256
    }" | jq '.choices[0].message.content'

❓ 常见问题
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Q: API 服务在哪里运行？
  A: http://localhost:8888

  Q: 如何查看 API 文档？
  A: 打开浏览器访问 http://localhost:8888/docs

  Q: 支持哪些编程语言？
  A: 任何可以发送 HTTP 请求的语言都支持（Python、JS、Go、Java等）

  Q: 如何启用性能优化？
  A: 设置环境变量后重启服务（见上方 ⚡ 性能优化 部分）

  Q: API 响应格式是什么？
  A: OpenAI 兼容格式，与 ChatGPT API 相同

  Q: 支持流式响应吗？
  A: 是的，设置 "stream": true 即可

  Q: 并发请求如何处理？
  A: 请求按队列顺序处理，暂不支持真正的并发

🐛 故障排查
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  问题: 连接被拒绝 (Connection refused)
  ───────────────────────────────────────────────────────────────────────────
  解决方案:
    1. 检查服务是否运行: ps aux | grep neurx_api
    2. 检查端口: netstat -tuln | grep 8888
    3. 重启服务: pkill -f neurx_api && sleep 2 && API_PORT=8888 python3 ...

  问题: 推理超时 (Timeout)
  ───────────────────────────────────────────────────────────────────────────
  解决方案:
    1. 减少 max_tokens: --max-tokens 50
    2. 检查系统资源: free -h && df -h
    3. 禁用优化: unset NEURX_KV_CACHE_ENABLED

  问题: 模型加载失败
  ───────────────────────────────────────────────────────────────────────────
  解决方案:
    1. 验证模型存在: ls -lh /app/shuwen/model/Qwen2.5-0.5B-Instruct/
    2. 设置正确路径: export NEURX_MODEL_PATH=/app/shuwen/model/...
    3. 检查权限: chmod -R 755 /app/shuwen/model/
